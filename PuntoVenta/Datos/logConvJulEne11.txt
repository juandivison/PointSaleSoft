CREATE PROCEDURE PROC_BALANCE_TOTAL (
  NUMERO_RECIBO INTEGER
) RETURNS (
  CANT_CUOTAS INTEGER,
  MONTO_CAPITAL_INICIAL NUMERIC(15, 2),
  BALANCE_ACT_PROY NUMERIC(15, 2),
  IDPRESTAMO INTEGER,
  CUOTAS INTEGER,
  INTERES NUMERIC(15, 2),
  BALANCEACT NUMERIC(15, 2),
  USAFACTOR SMALLINT,
  ABONO_A_CUOTAS NUMERIC(15, 2),
  CPIC_PAGADOS NUMERIC(15, 2),
  BALANCEANT NUMERIC(15, 2),
  SOLO_INTERES_PAGADO SMALLINT,
  MONTO_CAPITAL_PAGADO NUMERIC(15, 2),
  BALANCE_ANT NUMERIC(15, 2),
  TOTAL_ACUM_PAGOS NUMERIC(15, 2)
) AS    
  Declare Variable Sum_total_abonado Numeric(15,2);
  Declare Variable Sum_total_cpic_pagados Numeric(15,2);
  Declare Variable InteresCldo smallint;
  Declare Variable Sum_total_intereses Numeric(15,2);
  Declare Variable Sum_total_capitalApagar Numeric(15,2);
  Declare Variable CantC integer;
begin
  Sum_total_intereses = 0;
  CantC = 0;

  Select interes_cldo from conf_recibo
  into :InteresCldo;

  Select id_prestamo, balance_act, abono_a_cuotas,
         cpic_pagados, solo_interes_pagado,
         monto_capital_pagado, balance_ant
  From RECIBO
  Where Numero=:numero_recibo
  Into :idPrestamo, :balanceact,
  :abono_a_cuotas, :cpic_pagados, :SOLO_INTERES_PAGADO, :monto_capital_pagado, :balance_ant;

  Select cant_cuotas From prestamo
  Where serie =:idPrestamo
  INTO
  :CantC;

  if ( CantC > 1 ) then
  begin
    Select Sum(abono_a_cuotas), Sum(cpic_pagados), Sum(abono_a_cuotas)
    From RECIBO
    Where Recibo.id_prestamo =:idPrestamo
    and Recibo.Numero <=:numero_recibo
    Into
    :Sum_total_abonado, :Sum_total_cpic_pagados, :Total_acum_pagos;
  end else
  begin
    Select Sum(abono_a_cuotas), Sum(cpic_pagados), Sum(abono_a_cuotas)
    From RECIBO
    Where Recibo.id_prestamo =:idPrestamo
    and Recibo.Numero =:numero_recibo
    Into
    :Sum_total_abonado, :Sum_total_cpic_pagados, :Total_acum_pagos;
  end

  Select Count(*) From pagos_prestamo
  Where numero =:idPrestamo
  And pagos_prestamo.status in ('A')
  Into :cuotas;

  if (InteresCldo = 1) then
  begin
    Select sum(monto_interes_apagar)+sum(monto_pendiente), sum(monto_capital_Apagar)+sum(monto_pendiente_interes) From Pagos_Prestamo
    Where Pagos_Prestamo.numero =:idPrestamo and
          Pagos_Prestamo.status in ("A")
    Into :Sum_total_intereses, :Sum_total_capitalApagar;

    if (Sum_total_capitalApagar is null) then
    Sum_total_capitalApagar = 0;

    if (Sum_total_intereses is null) then
    begin
      Sum_total_intereses = 0;
    end
  end
  Select porc_interes, usa_factor, monto_capital, cant_cuotas From prestamo
  Where serie =:idPrestamo
  And status = 'A'
  Into
  :interes, :usafactor, :monto_capital_inicial, :cant_cuotas;

  if (:usafactor = 1) then
  begin
    if (InteresCldo = 1) then
    begin
      monto_capital_inicial =:monto_capital_inicial  + (:monto_capital_inicial * ((:interes/30) / 100)) * :cant_cuotas;
    end  else
    begin
      monto_capital_inicial =:monto_capital_inicial  + Sum_total_intereses;
    end
  end else
  begin
    if (InteresCldo = 1) then
    begin
      monto_capital_inicial =:monto_capital_inicial + (:monto_capital_inicial * (:interes / 100)) * :cant_cuotas;
    end else
    begin
      monto_capital_inicial =:monto_capital_inicial + Sum_total_intereses;
    end
  end

  if (:Sum_total_cpic_pagados Is Null) then
  Sum_total_cpic_pagados=0;

  if (:Sum_total_abonado Is Null) then
  Sum_total_abonado =0;

  balance_act_proy=:monto_capital_inicial - (:Sum_total_abonado); /* + :Sum_total_cpic_pagados); */
  if (abono_a_cuotas Is Null) then
  abono_a_cuotas = 0;

  if (cpic_pagados Is Null) then
  cpic_pagados = 0;

   if (SOLO_INTERES_PAGADO = 1) then
        Balanceant =:balance_act_proy + :Sum_total_abonado; /* + :cpic_pagados; */
   else Balanceant =:balance_act_proy + :Sum_total_abonado; /* + :cpic_pagados; */
   if (InteresCldo = 0) then
   begin
     balance_act_proy =:balanceact;
     Balanceant =:balance_ant;
   end
  Suspend;
end
